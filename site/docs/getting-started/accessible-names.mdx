The package sets the roles, the tab stop, the focus and the keys; it sets no text and no
`aria-label`. Every name a screen reader reads is yours: the grid's, each event's, each button's,
and the words of each announcement.

## Name the grid and the buttons

A grid needs a name, and an icon button needs one too:

<Framework name="react">

```tsx
<Calendar.DayGrid aria-label="November 2026">{/* … */}</Calendar.DayGrid>

<Calendar.Previous aria-label="Previous month">
    <ChevronLeft aria-hidden />
</Calendar.Previous>
<Calendar.ViewButton view="week">Week</Calendar.ViewButton>
```

</Framework>

<Framework name="vue">

```vue
<Calendar.DayGrid aria-label="November 2026"><!-- … --></Calendar.DayGrid>

<Calendar.Previous aria-label="Previous month">
    <ChevronLeft aria-hidden="true" />
</Calendar.Previous>
<Calendar.ViewButton view="week">Week</Calendar.ViewButton>
```

</Framework>

<Framework name="angular">

```html
<div calendarDayGrid aria-label="November 2026"><!-- … --></div>

<button calendarPrevious aria-label="Previous month">
    <svg lucideChevronLeft aria-hidden="true"></svg>
</button>
<button calendarViewButton="week">Week</button>
```

</Framework>

`Calendar.ViewButton` sets `aria-pressed` for the view shown; `Calendar.Previous`, `Today` and
`Next` are `disabled` (and carry `data-disabled`) when the model cannot run their command.

## Name each event

An event's part is a `listitem` inside its day's `list`. Its accessible name is its content, or an
`aria-label` you give it: say what a sighted reader sees, the title and the times.

<Framework name="react">

```tsx
function TimedEvent({ segment }: { segment: TimeSegment }) {
    const { model } = useCalendar();
    const event = model.get("event-by", { id: segment.eventId });
    const occurrence = model.get("occurrence-by", { id: segment.occurrenceId });
    const times = occurrence ? timesText(occurrence.start, occurrence.end) : "";
    return (
        <Calendar.TimeEvent segment={segment} aria-label={`${event?.title}, ${times}`}>
            <span>{event?.title}</span>
            <span aria-hidden>{times}</span>
        </Calendar.TimeEvent>
    );
}

function timesText(start: string, end: string): string {
    return new Intl.DateTimeFormat("en-US", { hour: "numeric", minute: "2-digit" })
        .formatRange(Date.parse(start), Date.parse(end));
}
```

</Framework>

<Framework name="vue">

```vue title="timed-event.vue"
<script setup lang="ts">
import { Calendar, type TimeSegment, useCalendar } from "@fragiola/calendar-vue";

const props = defineProps<{ segment: TimeSegment }>();
const { model } = useCalendar();

function title() {
    return model.get("event-by", { id: props.segment.eventId })?.title;
}

function times() {
    const occurrence = model.get("occurrence-by", { id: props.segment.occurrenceId });
    return occurrence ? timesText(occurrence.start, occurrence.end) : "";
}

function timesText(start: string, end: string): string {
    return new Intl.DateTimeFormat("en-US", { hour: "numeric", minute: "2-digit" })
        .formatRange(Date.parse(start), Date.parse(end));
}
</script>

<template>
    <Calendar.TimeEvent :segment="segment" :aria-label="`${title()}, ${times()}`">
        <span>{{ title() }}</span>
        <span aria-hidden="true">{{ times() }}</span>
    </Calendar.TimeEvent>
</template>
```

</Framework>

<Framework name="angular">

```ts title="timed-event.ts"
import { Component, computed } from "@angular/core";
import { injectCalendar, injectTimeEvent } from "@fragiola/calendar-angular";

// <div [calendarTimeEvent]="segment" appTimedEvent></div>
@Component({
    selector: "[appTimedEvent]",
    host: { "[attr.aria-label]": "title() + ', ' + times()" },
    template: `
        <span>{{ title() }}</span>
        <span aria-hidden="true">{{ times() }}</span>
    `,
})
export class TimedEvent {
    private readonly event = injectTimeEvent();
    private readonly model = injectCalendar().model;
    protected readonly title = computed(() => {
        const state = this.event.state();
        return state ? this.model().get("event-by", { id: state.eventId })?.title : "";
    });
    protected readonly times = computed(() => {
        const state = this.event.state();
        const occurrence = state && this.model().get("occurrence-by", { id: state.occurrenceId });
        return occurrence ? timesText(occurrence.start, occurrence.end) : "";
    });
}

function timesText(start: string, end: string): string {
    return new Intl.DateTimeFormat("en-US", { hour: "numeric", minute: "2-digit" })
        .formatRange(Date.parse(start), Date.parse(end));
}
```

</Framework>

## Say the dates

A day cell of a month shows "26"; a screen reader needs "Thursday, November 26". The date is data
on every day, column and slot (`data-date`, and the `day.date` your children function or slot
receives):
write it as visually hidden text.

<Framework name="react">

```tsx
<Calendar.Day day={day}>
    <span aria-hidden>{Number(day.date.slice(8))}</span>
    <span className="visually-hidden">{dayInWords(day.date)}</span>
    <Calendar.DayEvents>{(segment) => <EventSegment segment={segment} />}</Calendar.DayEvents>
</Calendar.Day>
```

</Framework>

<Framework name="vue">

```vue
<Calendar.Day :day="day">
    <span aria-hidden="true">{{ Number(day.date.slice(8)) }}</span>
    <span class="visually-hidden">{{ dayInWords(day.date) }}</span>
    <Calendar.DayEvents v-slot="{ segment }">
        <EventSegment :segment="segment" />
    </Calendar.DayEvents>
</Calendar.Day>
```

</Framework>

<Framework name="angular">

```html
<div [calendarDay]="day">
    <span aria-hidden="true">{{ +day.date.slice(8) }}</span>
    <span class="visually-hidden">{{ dayInWords(day.date) }}</span>
    <div calendarDayEvents>
        <ng-template calendarDayEventsItem let-segment>
            <div [calendarSegment]="segment" appEventSegment></div>
        </ng-template>
    </div>
</div>
```

</Framework>

```ts
function dayInWords(date: string): string {
    const [year = 1970, month = 1, day = 1] = date.split("-").map(Number);
    return new Intl.DateTimeFormat("en-US", {
        weekday: "long",
        month: "long",
        day: "numeric",
        timeZone: "UTC",
    }).format(Date.UTC(year, month - 1, day));
}
```

## The keyboard model

There is no WAI-ARIA pattern for an event calendar; the calendar combines two. The month and the
time grid are an [APG grid](https://www.w3.org/WAI/ARIA/apg/patterns/grid/) with one tab stop and
a roving focus over the dates, with the date picker's keys, and each date's events are a list of
focusable items.

| keys | on a date (a day, or a time-grid slot) | on an event |
| --- | --- | --- |
| Arrows | move a day (Up and Down: a week in a month, a slot in a time grid) | Up and Down: the day's previous or next event |
| Home, End | the first or last day of the week | |
| PageUp, PageDown | the previous or next range (`view.previous`, `view.next`), announced (`range`) | |
| Shift + PageUp, PageDown | a year (month), a week (time grids) | |
| Enter (or F2 on a date) | into the date's events (a slot: the events starting in it) | Enter: `event-click` |
| Escape | | back to its date |
| Tab | leaves the grid | leaves the grid |

A day's "+N more" link (`Calendar.More`) is the last item of its events: Down from the day's last
event reaches it, and Enter on it is the engine's `more-click`. With the features installed,
Shift and the arrows on a date select a range ([Selecting ranges](/docs/guides/selecting-ranges)),
and Space on an event grabs it to move it with the arrows and resize it with Shift and the arrows
([Keyboard](/docs/guides/keyboard)).

The keys follow the reading direction: in a right-to-left calendar, Left moves to the next day.

## Word the announcements

A drag by the keyboard, a selection grown with Shift, or a range a key showed is announced as
data: `onAnnounce` (`@announce` in Vue, `(announce)` in Angular) receives what happened (`grab`,
`drag-over`, `drop`, `drag-cancel`, `select-extend`, `range`) with the occurrence and its times, or
the range and the focused cell. The cells are keyed by position, so PageUp and PageDown may keep
focus on the same element with a new date, and no focus event tells a screen reader: the `range`
announcement does, once your polite live region says it. Write the words into a live region of
your own:

<Framework name="react">

```tsx
const [message, setMessage] = useState("");

<Calendar.Root
    model={model}
    onAnnounce={({ announcement }) => {
        if (announcement.kind === "grab") setMessage("Grabbed. Arrows move it, Enter drops it.");
        if (announcement.kind === "drag-cancel") setMessage("Left where it was.");
        if (announcement.kind === "range") setMessage(`From ${announcement.range.start}, on ${announcement.focused.date}.`);
    }}
>
    <div role="status" aria-live="polite">{message}</div>
    {/* … */}
</Calendar.Root>
```

</Framework>

<Framework name="vue">

```vue
<script setup lang="ts">
import type { Announcement } from "@fragiola/calendar-vue";
import { ref } from "vue";

const message = ref("");

function announce({ announcement }: { announcement: Announcement }) {
    if (announcement.kind === "grab") message.value = "Grabbed. Arrows move it, Enter drops it.";
    if (announcement.kind === "drag-cancel") message.value = "Left where it was.";
    if (announcement.kind === "range") message.value = `From ${announcement.range.start}, on ${announcement.focused.date}.`;
}
</script>

<template>
    <Calendar.Root :model="model" @announce="announce">
        <div role="status" aria-live="polite">{{ message }}</div>
        <!-- … -->
    </Calendar.Root>
</template>
```

</Framework>

<Framework name="angular">

```ts
import { signal } from "@angular/core";
import type { CalendarEngineEventOf } from "@fragiola/calendar-angular";

// a field and a method of the component; the template:
// <div [calendarRoot]="model" (announce)="announce($event)">
//     <div role="status" aria-live="polite">{{ message() }}</div> …
// </div>
readonly message = signal("");

announce({ announcement }: CalendarEngineEventOf<"announce">) {
    if (announcement.kind === "grab") this.message.set("Grabbed. Arrows move it, Enter drops it.");
    if (announcement.kind === "drag-cancel") this.message.set("Left where it was.");
    if (announcement.kind === "range") this.message.set(`From ${announcement.range.start}, on ${announcement.focused.date}.`);
}
```

</Framework>
